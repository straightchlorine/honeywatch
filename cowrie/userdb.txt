# Honeywatch custom Cowrie UserDB.
# Format per line: username:uid:password
#   *   matches anything (accept)
#   !X  rejects literal X
#   /re/ treats X as a regex (do NOT ship a `/honeypot/i` reject - it is
#        the textbook scanner signature for upstream userdb.example)
#
# Goal: look like a poorly-secured CentOS box with a handful of weak
# credentials scanners expect to work. Accept/reject ratio mirrors what a
# real compromised box looks like to the bots that probe it.

# Weak root logins bots expect to work, minus the obvious ones we reject.
# Curated list instead of wildcard -- a real neglected box has a handful of
# weak passwords, not "literally everything works" (which is itself a tell
# when a scanner tries 100 credentials and gets 96 successes).
root:x:!root
root:x:!toor
root:x:!12345
root:x:!password
root:x:123456
root:x:admin
root:x:123456789
root:x:qwerty
root:x:abc123
root:x:letmein
root:x:changeme

# Common admin / service accounts
admin:x:admin
admin:x:admin123
admin:x:password
administrator:x:admin
administrator:x:password

# Distro default users (CentOS / cloud images)
centos:x:centos
centos:x:centos123

# Debian / Ubuntu default cloud images (attackers spray these too)
ubuntu:x:ubuntu
